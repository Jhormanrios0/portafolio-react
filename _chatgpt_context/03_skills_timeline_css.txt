export function initSkillsAnimation({ gsap, ScrollTrigger }) {
  const section = document.querySelector(".section--skills");
  if (!section) return () => {};

  const story = section.querySelector("[data-skills-story]");
  const summary = section.querySelector("[data-skills-summary]");
  const phases = gsap.utils.toArray(
    section.querySelectorAll("[data-skills-phase]")
  );
  const ambient = section.querySelector(".skills-story__ambient");

  if (!story || !summary || !phases.length) return () => {};

  const cleanup = [];

  const timelineChromeSelector = [
    ".timeline-rail",
    ".timeline-line",
    ".timeline-svg",
    ".timeline-path",
    ".timeline-path-progress",
    ".timeline-traveler-dot",
    ".timeline-section__marker",
  ].join(",");

  const getTimelineChrome = () =>
    gsap.utils.toArray(document.querySelectorAll(timelineChromeSelector));

  const hideTimelineChrome = () => {
    document.body.classList.add("skills-story-lock");

    gsap.to(getTimelineChrome(), {
      autoAlpha: 0,
      x: -18,
      scale: 0.98,
      duration: 0.28,
      ease: "power3.out",
      overwrite: true,
    });
  };

  const showTimelineChrome = () => {
    document.body.classList.remove("skills-story-lock");

    gsap.to(getTimelineChrome(), {
      autoAlpha: 1,
      x: 0,
      scale: 1,
      duration: 0.34,
      ease: "power3.out",
      overwrite: true,
    });
  };

  const forceSectionClean = () => {
    const card = section.querySelector(".section-card");
    if (!card) return;

    gsap.set(card, {
      autoAlpha: 1,
      x: 0,
      y: 0,
      scale: 1,
      rotation: 0,
      filter: "none",
      clearProps: "transform",
    });
  };

  const getDeltaFromTitle = (title, element) => {
    if (!title || !element) return { x: 0, y: 0 };

    const titleRect = title.getBoundingClientRect();
    const elementRect = element.getBoundingClientRect();

    const titleCenterX = titleRect.left + titleRect.width / 2;
    const titleCenterY = titleRect.top + titleRect.height / 2;

    const elementCenterX = elementRect.left + elementRect.width / 2;
    const elementCenterY = elementRect.top + elementRect.height / 2;

    return {
      x: titleCenterX - elementCenterX,
      y: titleCenterY - elementCenterY,
    };
  };

  const setBaseState = () => {
    forceSectionClean();

    gsap.set(summary, {
      autoAlpha: 0,
      y: 50,
      scale: 0.965,
      force3D: true,
    });

    phases.forEach((phase) => {
      const eyebrow = phase.querySelector("[data-skills-eyebrow]");
      const title = phase.querySelector("[data-skills-title]");
      const description = phase.querySelector("[data-skills-description]");
      const cards = gsap.utils.toArray(
        phase.querySelectorAll("[data-skills-card]")
      );

      gsap.set(phase, {
        autoAlpha: 0,
        pointerEvents: "none",
      });

      gsap.set([eyebrow, description].filter(Boolean), {
        autoAlpha: 0,
        y: 24,
        force3D: true,
      });

      gsap.set(title, {
        autoAlpha: 0,
        y: 130,
        scale: 1.75,
        letterSpacing: "-0.17em",
        transformOrigin: "50% 50%",
        force3D: true,
      });

      gsap.set(cards, {
        autoAlpha: 0,
        scale: 0.08,
        rotationX: -18,
        rotationY: 0,
        rotationZ: 0,
        transformOrigin: "50% 50%",
        force3D: true,
      });
    });

    if (ambient) {
      gsap.set(ambient, {
        autoAlpha: 0.68,
        x: 0,
        y: 0,
        scale: 1,
        force3D: true,
      });
    }
  };

  const animatePhaseIn = (tl, phase, position) => {
    const eyebrow = phase.querySelector("[data-skills-eyebrow]");
    const title = phase.querySelector("[data-skills-title]");
    const description = phase.querySelector("[data-skills-description]");
    const cards = gsap.utils.toArray(
      phase.querySelectorAll("[data-skills-card]")
    );

    tl.set(phase, { autoAlpha: 1, pointerEvents: "auto" }, position);

    // 1) titulo
    tl.fromTo(
      title,
      {
        autoAlpha: 0,
        y: 140,
        scale: 1.78,
        letterSpacing: "-0.18em",
      },
      {
        autoAlpha: 1,
        y: 0,
        scale: 1,
        letterSpacing: "-0.08em",
        duration: 0.95,
        ease: "power4.out",
      },
      position
    );

    // 2) eyebrow mas tarde
    tl.to(
      eyebrow,
      {
        autoAlpha: 1,
        y: 0,
        duration: 0.34,
        ease: "power3.out",
      },
      position + 0.62
    );

    // 3) descripcion aun mas tarde
    tl.to(
      description,
      {
        autoAlpha: 1,
        y: 0,
        duration: 0.42,
        ease: "power3.out",
      },
      position + 0.96
    );

    // 4) cards una por una, mucho mas separadas
    cards.forEach((card, index) => {
      const cardStart = position + 1.42 + index * 0.22;

      tl.fromTo(
        card,
        {
          autoAlpha: 0,
          x: () => getDeltaFromTitle(title, card).x,
          y: () => getDeltaFromTitle(title, card).y,
          scale: 0.08,
          rotationX: -18,
          rotationY: index % 2 === 0 ? -14 : 14,
          rotationZ: [-10, 8, -7, 6, -5, 5][index % 6],
        },
        {
          autoAlpha: 1,
          x: 0,
          y: 0,
          scale: 1,
          rotationX: 0,
          rotationY: 0,
          rotationZ: 0,
          duration: 0.6,
          ease: "power4.out",
        },
        cardStart
      );
    });

    const cardsEnd = position + 1.42 + cards.length * 0.22 + 0.7;

    // 5) hold mas largo
    tl.to(
      cards,
      {
        y: (index) => (index % 2 === 0 ? -3 : 3),
        duration: 0.22,
        stagger: {
          each: 0.01,
          from: "center",
        },
        ease: "sine.inOut",
      },
      cardsEnd + 0.18
    );

    tl.to(
      cards,
      {
        y: 0,
        duration: 0.22,
        stagger: {
          each: 0.01,
          from: "center",
        },
        ease: "sine.inOut",
      },
      cardsEnd + 0.42
    );

    return cardsEnd + 1.05;
  };

  const animatePhaseOut = (tl, phase, position) => {
    const eyebrow = phase.querySelector("[data-skills-eyebrow]");
    const title = phase.querySelector("[data-skills-title]");
    const description = phase.querySelector("[data-skills-description]");
    const cards = gsap.utils.toArray(
      phase.querySelectorAll("[data-skills-card]")
    );

    [...cards].reverse().forEach((card, reverseIndex) => {
      const index = cards.length - 1 - reverseIndex;

      tl.to(
        card,
        {
          autoAlpha: 0,
          x: () => getDeltaFromTitle(title, card).x,
          y: () => getDeltaFromTitle(title, card).y,
          scale: 0.1,
          rotationX: -12,
          rotationY: index % 2 === 0 ? -10 : 10,
          rotationZ: index % 2 === 0 ? -7 : 7,
          duration: 0.34,
          ease: "power3.in",
        },
        position + reverseIndex * 0.08
      );
    });

    tl.to(
      [eyebrow, description].filter(Boolean),
      {
        autoAlpha: 0,
        y: -18,
        duration: 0.28,
        ease: "power2.in",
      },
      position + 0.18
    );

    tl.to(
      title,
      {
        autoAlpha: 0,
        y: -100,
        scale: 0.78,
        letterSpacing: "-0.14em",
        duration: 0.44,
        ease: "power3.in",
      },
      position + 0.56
    );

    tl.set(phase, { pointerEvents: "none" }, position + 1.12);

    return position + 1.32;
  };

  setBaseState();

  const lockTrigger = ScrollTrigger.create({
    trigger: section,
    start: "top 94%",
    end: "bottom 10%",
    invalidateOnRefresh: true,
    onEnter: hideTimelineChrome,
    onEnterBack: hideTimelineChrome,
    onLeave: showTimelineChrome,
    onLeaveBack: showTimelineChrome,
  });

  cleanup.push(() => {
    lockTrigger.kill();
    showTimelineChrome();
  });

  const tl = gsap.timeline({
    defaults: { ease: "none" },
    scrollTrigger: {
      trigger: section,
      start: "top 86%",
      end: "bottom 12%",
      scrub: 1.15,
      invalidateOnRefresh: true,
      onRefreshInit: forceSectionClean,
      onEnter: hideTimelineChrome,
      onEnterBack: hideTimelineChrome,
      onLeave: showTimelineChrome,
      onLeaveBack: showTimelineChrome,
    },
  });

  // resumen mas lento
  tl.to(
    summary,
    {
      autoAlpha: 1,
      y: 0,
      scale: 1,
      duration: 0.85,
      ease: "power3.out",
    },
    0
  );

  tl.to(
    summary,
    {
      autoAlpha: 1,
      y: 0,
      scale: 1,
      duration: 1.1,
      ease: "none",
    },
    0.9
  );

  tl.to(
    summary,
    {
      autoAlpha: 0,
      y: -64,
      scale: 0.965,
      duration: 0.56,
      ease: "power2.in",
    },
    2.08
  );

  let cursor = 3.1;

  phases.forEach((phase, index) => {
    const phaseEndVisible = animatePhaseIn(tl, phase, cursor);
    const phaseOutEnd = animatePhaseOut(tl, phase, phaseEndVisible + 0.65);

    cursor = phaseOutEnd + 0.9;

    if (ambient) {
      const positions = [
        { x: -120, y: 46, scale: 0.94, alpha: 0.58 },
        { x: 130, y: -52, scale: 1.05, alpha: 0.66 },
        { x: -82, y: 70, scale: 0.92, alpha: 0.52 },
        { x: 40, y: -34, scale: 0.86, alpha: 0.36 },
      ];

      const next = positions[index];

      tl.to(
        ambient,
        {
          x: next.x,
          y: next.y,
          scale: next.scale,
          autoAlpha: next.alpha,
          duration: 1.4,
          ease: "power2.out",
        },
        cursor - 2.4
      );
    }
  });

  tl.to({}, { duration: 0.8 });

  cleanup.push(() => tl.kill());

  const refresh = () => {
    forceSectionClean();
    ScrollTrigger.refresh();
  };

  requestAnimationFrame(refresh);
  window.addEventListener("load", refresh, { once: true });

  cleanup.push(() => {
    window.removeEventListener("load", refresh);
  });

  return () => {
    cleanup.forEach((fn) => fn());
  };
}
